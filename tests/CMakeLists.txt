cmake_minimum_required(VERSION 3.21)
project(STBReloadTests LANGUAGES CXX)
enable_testing()
option(STB_TEST_ASAN "Run the offline suite with AddressSanitizer" OFF)
find_package(Threads REQUIRED)
add_executable(stb_reload_tests reload_tests.cpp ../src/HudRuntime.cpp)
target_compile_features(stb_reload_tests PRIVATE cxx_std_20)
target_include_directories(stb_reload_tests PRIVATE . ../src)
target_link_libraries(stb_reload_tests PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(stb_reload_tests PRIVATE /W4 /WX /FIEngineStub.h)
    if(STB_TEST_ASAN)
        target_compile_options(stb_reload_tests PRIVATE /fsanitize=address /Zi)
        target_link_options(stb_reload_tests PRIVATE /INCREMENTAL:NO)
    endif()
else()
    target_compile_options(stb_reload_tests PRIVATE -Wall -Wextra -Werror -include EngineStub.h)
    if(STB_TEST_ASAN)
        target_compile_options(stb_reload_tests PRIVATE -fsanitize=address)
        target_link_options(stb_reload_tests PRIVATE -fsanitize=address)
    endif()
endif()
add_test(NAME reload_lifecycle COMMAND stb_reload_tests)
